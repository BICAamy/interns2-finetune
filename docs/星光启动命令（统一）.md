### 先进入工作路径
```bash
cd /UNICOMFS/shskw43_1/.bihu/interns2-agent-robot/interns2-finetune
```

### 统一启动
```bash
cd ~/LSL/interns2-dev-bundle

export CUDA_HOME=~/LSL/conda_envs/cuda128
export CUDA_PATH="$CUDA_HOME"
export PATH="$CUDA_HOME/bin:$PATH"

export NCCL_P2P_DISABLE=1
export NCCL_IB_DISABLE=1
export NCCL_CUMEM_HOST_ENABLE=0
export NCCL_SHM_DISABLE=0

INFERENCE_CUDA_VISIBLE_DEVICES=0,1,2,3 \
INFERENCE_TP=4 \
./scripts/start_all.sh
```
### 去本地主机上端口转发
```bash
ssh -N \
  -o ExitOnForwardFailure=yes \
  -L 18000:127.0.0.1:8000 \
  5090
```
这里的interns2-robot-new对应Host，非HostName

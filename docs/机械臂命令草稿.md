# 1. 启动服务器
```bash
cd /home/ubuntu/LSL/interns2-dev-bundle/app

export GATEWAY_AUTH_SECRET_FILE="$PWD/configs/gateway-auth.local"
export GATEWAY_EXPECTED_ID=mac-huayan-ph590320009
export ROBOT_REAL_MIRROR=1

./scripts/services/start_all.sh \
  --robot-mode real \
  --real-config configs/robot-real.local.yaml
```

## 2. 建立主机与服务器隧道
```bash
# 将mac端口18001映射到服务器（5090）的端口18011上
# 18000：网页
# 18001：Mac 网关连接服务器 robot-runtime
ssh -N \
  -o ExitOnForwardFailure=yes \
  -L 18000:127.0.0.1:8000 \
  -L 18001:127.0.0.1:8001 \
  5090
```

# 3. 启动网关
```bash
# 启动网关
EDGE_PYTHON=/Users/bigcat/Documents/internS/venv/course/bin/python \
./scripts/edge/start_gateway.sh \
  --real-config configs/robot-real.local.yaml \
  --connect-real \
  --server-url ws://127.0.0.1:18001/v1/gateway/connect \
  --secret-file configs/gateway-auth.local \
  --gateway-id mac-huayan-ph590320009 \
  --datasheet-byte-order little \
  --audit-path logs/edge_gateway.log
```
